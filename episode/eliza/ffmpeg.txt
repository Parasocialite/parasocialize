# The episode as an mp4. This is the only step that needs ffmpeg:
#     winget install ffmpeg        (or:  choco install ffmpeg)
#
cd C:\Users\paras\Documents\Default Project\parasocialize\episode\eliza
ffmpeg -y -f concat -safe 0 -i "frames.txt" -i narration.wav \
  -vsync vfr -c:v libx264 -pix_fmt yuv420p -crf 20 -preset slow \
  -c:a aac -b:a 128k -shortest -movflags +faststart "eliza.mp4"
#
# horizontal master for youtube, same frames, same audio:
ffmpeg -y -f concat -safe 0 -i "frames.txt" -i narration.wav \
  -vsync vfr -vf "scale=1920:1080:force_original_aspect_ratio=increase,crop=1920:1080" -c:v libx264 -pix_fmt yuv420p -crf 20 \
  -c:a aac -b:a 128k -shortest "eliza-16x9.mp4"
#
